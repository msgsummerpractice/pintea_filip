# User List Implementation Plan

> **Execution:** Use subagent-driven development to implement this plan task-by-task.

**Goal:** Add an admin-only user-list API and a React user-list page that shows all accounts with role labels, passenger-linked case counts, and placeholder actions.

### Task 1: Add Admin-Only User List API

**Files:**
- Modify: `backend/apps/cases/api/views.py`
- Modify: `backend/apps/cases/api/serializers.py`
- Modify: `backend/apps/cases/api/urls.py`

**Requirements:**
- Expose a read-only endpoint that returns all auth users.
- Restrict the endpoint to authenticated system admins.
- Return `id`, `name`, `email`, `role`, `assignedCaseCount`, and placeholder action metadata.
- Derive roles with the required precedence: `System Admin`, `Colleague`, then `Passenger`.
- Compute assigned case counts from `Passenger.user -> Passenger.cases` without N+1 queries.

**Implementation:**

```python
from django.contrib.auth import get_user_model
from django.db.models import Case as DbCase
from django.db.models import CharField, Count, Q, Value, When
from rest_framework.permissions import IsAdminUser


class UserListItemSerializer(serializers.Serializer):
    id = serializers.IntegerField()
    name = serializers.CharField()
    email = serializers.EmailField()
    role = serializers.CharField()
    assignedCaseCount = serializers.IntegerField()
    actions = serializers.DictField(child=serializers.BooleanField())


class AdminUserListView(APIView):
    permission_classes = [IsAdminUser]

    def get(self, request) -> Response:
        user_model = get_user_model()
        users = (
            user_model.objects.annotate(
                assigned_case_count=Count("passengers__cases", distinct=True),
                passenger_count=Count("passengers", distinct=True),
                derived_role=DbCase(
                    When(is_superuser=True, then=Value("System Admin")),
                    When(is_staff=True, then=Value("Colleague")),
                    When(passengers__isnull=False, then=Value("Passenger")),
                    default=Value("Passenger"),
                    output_field=CharField(),
                ),
            )
            .order_by("email", "id")
            .distinct()
        )

        payload = [
            {
                "id": user.id,
                "name": (f"{user.first_name} {user.last_name}".strip() or user.email),
                "email": user.email,
                "role": user.derived_role,
                "assignedCaseCount": user.assigned_case_count,
                "actions": {"edit": False, "delete": False},
            }
            for user in users
        ]
        serializer = UserListItemSerializer(payload, many=True)
        return Response({"results": serializer.data}, status=status.HTTP_200_OK)
```

```ts
  id: number;
  name: string;
  email: string;

```

```ts
import { requestJson } from "../../lib/http";
import type { UserListRow } from "./types";

interface UserListResponse {
  results: Array<{
    id: number;
    name: string;
    email: string;
    role: UserListRow["role"];
    assignedCaseCount: number;
    actions: {
      edit: boolean;
      delete: boolean;
    };
  }>;
}

export async function fetchUserList(): Promise<UserListRow[]> {
  const response = await requestJson<UserListResponse>("/users/");
  return response.results;
}
```

```tsx
import { useEffect, useState } from "react";

import { HttpError } from "../../../lib/http";
import { fetchUserList } from "../api";
import type { UserListRow } from "../types";

export function UserListPage() {
  const [rows, setRows] = useState<UserListRow[] | null>(null);
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    let cancelled = false;

    fetchUserList()
      .then((result) => {
        if (!cancelled) {
          setRows(result);
        }
      })
      .catch((reason: unknown) => {
        if (cancelled) {
          return;
        }

        if (reason instanceof HttpError && (reason.status === 401 || reason.status === 403)) {
          setError("You do not have access to the user list.");
          return;
        }

        setError("Unable to load users right now.");
      });

    return () => {
      cancelled = true;
    };
  }, []);

  return (
    <main className="user-list-page">
      <section className="user-list-frame" aria-labelledby="user-list-title">
        <header className="user-list-header">
          <p className="eyebrow">System administration</p>
          <h1 id="user-list-title">User List</h1>
          <p>Review all system users, their role, and passenger-linked case counts.</p>
        </header>

        {rows === null && error === null ? <p role="status">Loading users...</p> : null}
        {error ? <div className="notice-banner notice-banner-error" role="alert">{error}</div> : null}
        {rows && rows.length === 0 ? <div className="notice-banner" role="status">No users found.</div> : null}
        {rows && rows.length > 0 ? (
          <div className="user-list-table-shell">
            <table>
              <thead>
                <tr>
                  <th>Name</th>
                  <th>E-Mail</th>
                  <th>Role</th>
                  <th>Assigned Cases</th>
                  <th>Actions</th>
                </tr>
              </thead>
              <tbody>
                {rows.map((row) => (
                  <tr key={row.id}>
                    <td>{row.name}</td>
                    <td>{row.email}</td>
                    <td>{row.role}</td>
                    <td>{row.assignedCaseCount}</td>
                    <td>
                      <div className="user-list-actions">
                        <button type="button" disabled aria-label={`Edit ${row.email}`}>Edit</button>
                        <button type="button" disabled aria-label={`Delete ${row.email}`}>Delete</button>
                      </div>
                    </td>
                  </tr>
                ))}
              </tbody>
            </table>
          </div>
        ) : null}
      </section>
    </main>
  );
}
```
